\setlist[enumerate]{topsep=0pt, itemsep=2pt, parsep=0pt}

\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=*, itemsep=3pt]

\item \textbf{(5 points)} There is no operating system on a bare-metal microcontroller to set up memory before your code runs -- the microcontroller's own startup code has to do that job itself.

\begin{enumerate}[label=(\alph*), leftmargin=*, itemsep=3pt]
\item \textbf{(2 points)} Which of the following correctly orders the sequence of events that take place on a generic Cortex-M microcontroller between power-up and the execution of the first instruction in \texttt{main()}?

\begin{center}
\begin{tabular}{@{}rl@{}}
\textbf{(A)} & SP load $\rightarrow$ reset-vector load $\rightarrow$ \texttt{main()} $\rightarrow$ \texttt{.data} copy $\rightarrow$ \texttt{.bss} zero $\rightarrow$ early setup \\
\textbf{(B)} & SP load $\rightarrow$ reset-vector load $\rightarrow$ early setup $\rightarrow$ \texttt{.data} copy $\rightarrow$ \texttt{.bss} zero $\rightarrow$ \texttt{main()} \\
\textbf{(C)} & early setup $\rightarrow$ SP load $\rightarrow$ reset-vector load $\rightarrow$ \texttt{.bss} zero $\rightarrow$ \texttt{.data} copy $\rightarrow$ \texttt{main()} \\
\textbf{(D)} & SP load $\rightarrow$ reset-vector load $\rightarrow$ \texttt{.data} copy $\rightarrow$ \texttt{.bss} zero $\rightarrow$ early setup $\rightarrow$ \texttt{main()} \\
\end{tabular}
\end{center}

\item \textbf{(2 points)} A global variable is declared \texttt{int sensorGain = 5;}. If the startup code's copy of the \texttt{.data} section from flash into SRAM were skipped, what value would \texttt{sensorGain} actually have the first time \texttt{main()} reads it?

\mcfour{\textbf{(A)} 5 (baked into the machine code)}{\textbf{(B)} 0 (SRAM defaults to zero)}{\textbf{(C)} Garbage already sitting in SRAM}{\textbf{(D)} The program fails to compile}

\item \textbf{(1 point)} A global variable is declared \texttt{static int sampleCount;} with no explicit initializer. If the \texttt{.bss} zero-initialization step were skipped at startup, what value would \texttt{sampleCount} have the first time \texttt{main()} reads it?

\mcfour{\textbf{(A)} 0 (C guarantees it, no hardware needed)}{\textbf{(B)} Garbage already sitting in SRAM}{\textbf{(C)} 5, same as \texttt{sensorGain}}{\textbf{(D)} The program fails to compile}
\end{enumerate}

\item \textbf{(5 points)} A bootloader and the application it manages are built as two separate programs that share one physical flash chip, as shown below (addresses are representative, not to scale).

\vspace{-4pt}
\flashMapDiagram
\vspace{-6pt}

\begin{enumerate}[label=(\alph*), leftmargin=*, itemsep=3pt]
\item \textbf{(1 point)} At reset, hardware always fetches the initial stack pointer and reset vector from the same fixed base address of flash (0x0800\_0000 above). Based on this, which program must be placed at that fixed address?

\mcfour{\textbf{(A)} The application}{\textbf{(B)} The bootloader}{\textbf{(C)} Either one -- it does not matter}{\textbf{(D)} A separate ``reset program''}

\item \textbf{(2 points)} The bootloader jumps to the application's reset handler without ever changing the processor's vector-table pointer. Immediately afterward, an interrupt fires. Whose interrupt handler actually executes?

\mcfour{\textbf{(A)} The application's}{\textbf{(B)} The bootloader's}{\textbf{(C)} Neither -- a hard fault, no handler at all}{\textbf{(D)} Whichever is closest in flash}

\item \textbf{(1 point)} What is the one action the application's startup code must take so that interrupts are correctly routed to its own handlers instead of the bootloader's?

\mcfour{\textbf{(A)} Recompile the bootloader without a vector table}{\textbf{(B)} Update the vector-table pointer to the application's own table}{\textbf{(C)} Move the application's code to 0x0800\_0000}{\textbf{(D)} Disable all interrupts permanently}

\item \textbf{(1 point)} Why must the bootloader and application be built as two separate projects, each with its own linker script defining a non-overlapping flash region?

\mcfour{\textbf{(A)} So each can be (re)flashed independently}{\textbf{(B)} One linker script cannot produce more than one vector table}{\textbf{(C)} They would otherwise be forced to share one stack pointer}{\textbf{(D)} It is not required -- only a style preference}
\end{enumerate}

\end{enumerate}
